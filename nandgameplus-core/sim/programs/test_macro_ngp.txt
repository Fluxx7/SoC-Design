INCLUDE stack_macros_ngp.txt
INCLUDE number_macros_ngp.txt

DEFINE addr 0x6001
DEFINE dispAddr 0x6002

init.stack
llimm dispAddr $a
llimm 0x4000 $a*

main:

call waitSync 0
pop.reg $d
$a = end
$d == $zr ?: $a
call render 0
$d = 0x20
llimm ispAddr $a
$a* = $a* + $d
$a = main
@: $a

function waitSync 1

LABEL wait_sync:
llimm addr $a
$d = $a*
$a = 2
$d = $d & $a
push.reg $d
pop.local 0

loop_sync:
llimm addr $a
$d = $a*
$a = 2
$d = $d & $a
push.reg $d
push.local 0
pop.reg $d
$a = loop_sync
$sp = $sp - 1
$d == $sp* ?: $a


llimm addr $d
$a = 1
$sp* = $d* & $a
$sp = $sp + 1
return

function render 1
push.value 0
pop.local 0

render_loop:
call waitSync 0
pop.reg $d
llimm dispAddr $a
$a = $a*
$d = $a* + $d
$a* = $a* + $d
push.local 0
push.value 1
add
pop.local 0
$a = 16
$d = $a - $d
$a = render_return
$d == $zr ?: $a
goto render_loop

render_return:
return


end:
